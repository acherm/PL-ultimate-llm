code.plaid · added: 2026-02-06T12:00:00Z
state File {
  val filename;
  
  state OpenFile {
    method read() {
      "reading from file"
    }
    method close() {
      this <- ClosedFile;
    }
  }
  
  state ClosedFile {
    method open() {
      this <- OpenFile;
    }
  }
}

method main() {
  val f = new File { filename = "test.txt"; } with ClosedFile;
  f.open();
  f.read();
  f.close();
}
