FACTORIAL: PROCEDURE OPTIONS(MAIN);
  DECLARE N FIXED BINARY;
  DECLARE RESULT FIXED BINARY;
  DECLARE I FIXED BINARY;

  N = 5;
  RESULT = 1;

  DO I = 1 TO N;
    RESULT = RESULT * I;
  END;

  PUT SKIP LIST('FACTORIAL OF', N, 'IS', RESULT);

END FACTORIAL;
