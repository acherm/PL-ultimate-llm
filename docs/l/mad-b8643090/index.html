SAMPLE PROGRAM
TITLE AVERAGE
DIMENSION A(100), SUM = 0.0
READ N, A(1), A(2), ..., A(N)
FOR I = 1 STEP 1 UNTIL N DO
  SUM <- SUM + A(I)
AVERAGE = SUM / N
PRINT AVERAGE
FORM 8F10.3
