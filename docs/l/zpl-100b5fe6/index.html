program matmult;

config var
  n : integer = 100;

region
  R = [1..n, 1..n];

var
  A, B, C : [R] double;

procedure main();
begin
  A := [i,j] i + j;
  B := [i,j] i - j;
  C := [i,j] sum([k] A[i,k] * B[k,j]);
end;

begin
  main();
end.
