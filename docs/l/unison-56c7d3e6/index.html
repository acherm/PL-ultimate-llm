fib : Nat -> Nat
fib n =
  if n <= 1 then n
  else fib (n - 1) + fib (n - 2)

test> fib 0 == 0
test> fib 1 == 1
test> fib 2 == 1
test> fib 3 == 2
test> fib 4 == 3
test> fib 5 == 5
