# Bell State Preparation
# This program creates a maximally entangled Bell state between two qubits

DECLARE ro BIT[2]

# Initialize qubit 0 in superposition
H 0

# Entangle qubit 0 and qubit 1
CNOT 0 1

# Measure both qubits
MEASURE 0 ro[0]
MEASURE 1 ro[1]
