;;; Simple TAK test

(define (tak x y z)
  (if (not (< y x))
      z
    (tak (tak (- x 1) y z)
         (tak (- y 1) z x)
         (tak (- z 1) x y))))

(define (main . args)
  (let ((result (tak 18 12 6)))
    (if (= result 7)
        (print "passed")
      (print "failed"))
    result))
