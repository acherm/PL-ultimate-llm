;; Simple GOAL example - factorial function
(defun factorial ((n int)) int
  (if (= n 0)
      1
      (* n (factorial (- n 1)))
      )
  )

;; Calculate and print factorial of 5
(let ((result (factorial 5)))
  (format #t "Factorial of 5 is ~D~%" result)
  )
