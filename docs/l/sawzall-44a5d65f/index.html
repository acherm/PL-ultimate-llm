# Count the number of pages viewed by each user.
# Input is a stream of weblogs.
# Output is a table mapping user name to a count of pages.

# define the input record format
proto "weblogs.proto"
input_record: record Weblog.LogRecord

# define the output table
pages_by_user: table sum of int;

# for each record, increment the count for that user
emit pages_by_user[input_record.user] <- 1;
