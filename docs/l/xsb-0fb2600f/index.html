Fibonacci with Tabling

Origin · Download · manifest.json
code.P · added: 2026-02-06T15:30:00Z
% Fibonacci numbers using tabling
:- table fib/2.

fib(0, 0).
fib(1, 1).
fib(N, F) :-
    N > 1,
    N1 is N - 1,
    N2 is N - 2,
    fib(N1, F1),
    fib(N2, F2),
    F is F1 + F2.

% Query to compute fibonacci of 10
:- fib(10, F), writeln(F).
