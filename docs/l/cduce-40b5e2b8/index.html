type person = `person[ `name[ String ] `age[ Int ] ]

let hello_world() =
  let p = `person[ `name["Alice"] `age[30] ] in
  match p with
  | `person[ `name[n] `age[a] ] ->
      print "Name: "; print n; print "\n";
      print "Age: "; print (string_of a); print "\n"
