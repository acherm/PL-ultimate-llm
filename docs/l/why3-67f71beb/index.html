module Intro

  use import int.Int

  let even (n:int) : bool = n mod 2 = 0

  goal G1: forall n:int. even n <-> n mod 2 = 0
    intros n. unfold even. reflexivity. qed

  goal G2: forall n:int. even (n + 2) <-> even n
    intros n. unfold even. rewrite (plus_remainder n 2). ring. qed

end
