Program ReadFile
	// This record defines the layout of the input file
	Record InRec type BasicRecord
		custNo 	String(5);
		custName String(20);
		custAddr String(25);
		custCity String(15);
		custState String(2);
		custZip String(5);
		custPhone String(10);
	end

	// This record defines the layout of the output file
	Record OutRec type BasicRecord
		custNo 	String(5);
		custName String(20);
		custPhone String(10);
	end

	// File definitions
	inFile file InRec;
	outFile file OutRec;

	Function main()
		// Open the files
		open inFile for input;
		open outFile for output;

		// Read the first record
		get inFile into InRec;

		// Loop through the input file
		while (sysVar.ioStatus == 0)
			// Call the program to format the phone number
			call FormatPhone(InRec, OutRec);

			// Write the output record
			add outFile from OutRec;

			// Read the next record
			get inFile into InRec;
		end

		// Close the files
		close inFile;
		close outFile;
	end
end
