/*
 * Fibonacci sequence in ICI
 */
auto fib(n)
{
    if (n <= 1)
        return n;
    return fib(n - 1) + fib(n - 2);
}

auto i;
for (i = 0; i < 15; ++i)
    printf("%d ", fib(i));
printf("\n");
