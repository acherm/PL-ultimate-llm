PROC fibonacci(INT n):
  INT a := 0;
  INT b := 1;
  INT i := 0;
  WHILE i < n DO
    put(a);
    INT temp := a;
    a := b;
    b := temp + b;
    i := i + 1;
  END WHILE;
END PROC fibonacci;
