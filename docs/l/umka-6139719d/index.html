func factorial(n: int): int {
    if n <= 1 {
        return 1
    }
    return n * factorial(n - 1)
}

func main() {
    printf("Hello from Umka!\n")
    for i := 1; i <= 10; i++ {
        printf("Factorial of %d is %d\n", i, factorial(i))
    }
}
