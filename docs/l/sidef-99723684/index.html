func qsort(lst) {
    var len = lst.len
    len < 2 and return lst
    var pivot = lst.pop
    var lo = lst.grep { _ < pivot }
    var hi = lst.grep { _ >= pivot }
    return qsort(lo) + [pivot] + qsort(hi)
}

say qsort([3,1,4,1,5,9,2,6])
