class Counter
  var count

  def init()
    count := 0
  end

  def increment()
    count := count + 1
  end

  def getValue()
    return count
  end
end

var myCounter := new Counter()
myCounter.increment()
myCounter.increment()
var result := myCounter.getValue()
print(result)
