// Factorial function
global factorial = function(n)
{
  if (n <= 1)
  {
    return 1;
  }
  return n * factorial(n - 1);
};

// Calculate and print factorials
print("Factorial of 5 is ", factorial(5));
print("Factorial of 10 is ", factorial(10));
