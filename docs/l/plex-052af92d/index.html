/* Simple PLEX program example */
BLOCK EXAMPLE;
  BEGIN
    VAR X, Y, Z : INTEGER;

    X := 10;
    Y := 20;
    Z := X + Y;

    IF Z > 25 THEN
      SIGNAL MESSAGE("Result is greater than 25");
    ELSE
      SIGNAL MESSAGE("Result is 25 or less");
    FI;
  END EXAMPLE;
