use std.io

fn main() {
  print("Enter the first number: ")
  let a = read_int()
  print("Enter the second number: ")
  let b = read_int()
  print("Enter the operator (+, -, *, /): ")
  let op = read_str()
  if op == "+" {
    print("Result: ", a + b)
  } else if op == "-" {
    print("Result: ", a - b)
  } else if op == "*" {
    print("Result: ", a * b)
  } else if op == "/" {
    print("Result: ", a / b)
  } else {
    print("Invalid operator")
  }
}
