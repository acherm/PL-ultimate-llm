DEFINE PROCEDURE "FACTORIAL" [N]:
BLOCK 0: BEGIN
    CELL(0) <= 1;
    LOOP AT MOST N TIMES:
    BLOCK 1: BEGIN
        CELL(0) <= CELL(0) * (N - LOOP-COUNTER + 1);
    BLOCK 1: END;
    RETURN CELL(0);
BLOCK 0: END.
