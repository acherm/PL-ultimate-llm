PROCEDURE FACTORIAL(N);
    INTEGER N;
    BEGIN
        INTEGER RESULT;
        IF N <= 1 THEN
            RESULT := 1
        ELSE
            RESULT := N * FACTORIAL(N - 1);
        RETURN RESULT;
    END;

BEGIN
    INTEGER I;
    FOR I := 1 STEP 1 UNTIL 10 DO
        BEGIN
            PRINT('FACTORIAL OF', I, 'IS', FACTORIAL(I));
        END;
END.
