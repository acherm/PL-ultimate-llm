C     COMPUTE PAYROLL TOTALS
      DIMENSION HOURS(10), RATE(10), PAY(10)
      READ HOURS
      READ RATE
      DO 10 I = 1, 10
         PAY(I) = HOURS(I) * RATE(I)
   10 CONTINUE
      TOTAL = 0.0
      DO 20 I = 1, 10
         TOTAL = TOTAL + PAY(I)
   20 CONTINUE
      PRINT TOTAL
      END
