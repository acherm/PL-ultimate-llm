<ROUTINE PARSER ("AUX" ICNT WCNT CNT)
 <SET ICNT <GETP ,P-INBUF 1>>
 <SET WCNT 0>
 <SET CNT 2>
 <REPEAT ()
  <SET WCNT <+ .WCNT 1>>
  <COND (<G? .WCNT ,P-MAXOWORDS>
	 <RFALSE>)>
  <SET <GET ,P-LEXV .WCNT> <GETP ,P-INBUF .CNT>>
  <SET <GET ,P-LEXL .WCNT> <GETP ,P-INBUF <+ .CNT 1>>>
  <SET CNT <+ .CNT 2>>
  <COND (<EQUAL? .CNT <+ .ICNT 2>>
	 <SET <GET ,P-LEXV 0> .WCNT>
	 <RTRUE>)>>
 <SETG P-CONT 0>
