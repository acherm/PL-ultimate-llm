% Render Markdown blockquotes (used for callout boxes) as shaded, left-barred boxes.
\usepackage{mdframed}
\usepackage{xcolor}
\definecolor{cobg}{RGB}{244,247,251}
\definecolor{cobar}{RGB}{11,61,97}
\surroundwithmdframed[
  topline=false, bottomline=false, rightline=false, leftline=true,
  linewidth=3pt, linecolor=cobar, backgroundcolor=cobg,
  innerleftmargin=10pt, innerrightmargin=8pt,
  innertopmargin=6pt, innerbottommargin=6pt,
  skipabove=8pt, skipbelow=8pt]{quote}
