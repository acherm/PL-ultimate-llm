code.tbasic · added: 2025-11-05T20:17:53Z
Uses "Console"

Dim a As Double
Dim b As Double
Dim op As String

PrintL "Simple Calculator"
PrintL "Enter first number: " In KeyInput
Input a

PrintL "Enter operator (+, -, *, /): " In KeyInput
Input op

PrintL "Enter second number: " In KeyInput
Input b

Select Case op
  Case "+"
    PrintL "Result: " & a + b
  Case "-"
    PrintL "Result: " & a - b
  Case "*"
    PrintL "Result: " & a * b
  Case "/"
    If b <> 0 Then
      PrintL "Result: " & a / b
    Else
      PrintL "Error: Division by zero"
    End If
  Case Else
    PrintL "Invalid operator"
End Select

PrintL "Press any key to exit..."
Wait Key
