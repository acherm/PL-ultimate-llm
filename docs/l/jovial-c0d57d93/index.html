PROC HANOI(N,F,V,T)
BEGIN
ITEM N BYTE;
ITEM F BYTE;
ITEM V BYTE;
ITEM T BYTE;
IF N > 0
THEN
BEGIN
HANOI(N-1,F,T,V);
PRINT(F,T);
HANOI(N-1,V,F,T);
END
END

PROC PRINT(A,B)
BEGIN
ITEM A BYTE;
ITEM B BYTE;
TERM;
PRINT("MOVE DISK FROM ");
PRINT(A);
PRINT(" TO ");
PRINT(B);
END
