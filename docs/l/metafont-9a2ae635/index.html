Simple Metafont program to draw letter A

Origin · Download · manifest.json
code.mf · added: 2025-10-28T06:22:56Z
mode setup;

font_family := example;
font_size := 10pt#;

beginchar("A", 10u#, 10u#, 0);
  italicslant slant;
  italiccorrection ic;
  pickup pensquare scaled 1.2pt;
  z1 = (0,0); bot z1 = 0;
  z2 = (5u,10u); top z2 = h;
  z3 = (10u,0); bot z3 = 0;
  z11 = (2.5u,5u); z12 = (7.5u,5u);
  fill z1--z2--z3--cycle cutafter z11--z12;
  penpos1(0,0); penpos2(5u,0); penpos3(10u,0);
  penpos4(0,h); penpos5(5u,h); penpos6(10u,h);
  filldraw (z1l..z2l..z3l) {dir 90} .. {dir 90} (z3r..z2r..z1r);
endchar;

bye.
