// Factorial function in Prograph (textual representation)
// This represents a recursive factorial computation

method factorial(n)
  local inputs: n
  local outputs: result
  
  case n <= 1
    return 1
  otherwise
    temp := n - 1
    recursive_result := factorial(temp)
    result := n * recursive_result
    return result
  end case
end method

// Main execution
main()
  input := 5
  output := factorial(input)
  print(output)  // prints 120
end main
