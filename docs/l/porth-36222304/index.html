Fibonacci Sequence Calculator

include lib/io.porth

: fib { n -- fib(n) }
  dup 1 < if
    drop 0
  else
    dup 1 < if
      drop 1
    else
      tuck fib
      1 - fib
      +
    then
  then
;

main:
  10 fib .
;
