-- The Fibonacci sequence starts with 0 and 1 and
each subsequent number is the sum of the previous two.

module Fibonacci where

fibs = 0 : scanl (+) 1 fibs

main = do
    println (take 20 fibs)
    println $ "fib 30 = " ++ show (fibs !! 30)
