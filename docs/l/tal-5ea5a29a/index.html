PROC HELLO MAIN;
  BEGIN
    INT .FILENAME[0:11] := "HOMETERM";
    INT .BUFFER[0:79];
    INT WLEN;
    INT ERROR;
    
    CALL OPEN(FILENAME, WLEN, ERROR);
    
    STRING .S := @BUFFER;
    S ':=' "Hello, World!" -> @WLEN;
    
    CALL WRITE(FILENAME, BUFFER, WLEN, ERROR);
    CALL CLOSE(FILENAME);
  END;
