PROCEDURE fibonacci
(* Compute and display Fibonacci sequence *)
TYPE
  INTEGER : i, n, fib1, fib2, temp
BEGIN
  PRINT "How many Fibonacci numbers? ";
  INPUT n

  fib1 := 0
  fib2 := 1

  PRINT "Fibonacci sequence:"
  PRINT fib1

  IF n > 1 THEN
    PRINT fib2
  ENDIF

  FOR i := 3 TO n DO
    temp := fib1 + fib2
    PRINT temp
    fib1 := fib2
    fib2 := temp
  ENDFOR
END
