FACT: PROCEDURE OPTIONS(MAIN);
   DECLARE N FIXED;
   DECLARE RESULT FIXED;
   DECLARE I FIXED;

   N = 5;
   RESULT = 1;
   DO I = 1 TO N;
      RESULT = RESULT * I;
   END;
   PUT SKIP LIST('Factorial of', N, 'is', RESULT);
END FACT;
