func fib(n -> int) -> int
{
    if (n < 2)
    {
        n
    }
    else
    {
        fib(n - 1) + fib(n - 2)
    }
}

func main() -> int
{
    let i = 0;

    for (i = 0; i < 10; i = i + 1)
    {
        prints("fib(" + i + ") = " + fib(i) + "\n")
    };

    0
}
