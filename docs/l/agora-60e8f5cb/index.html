// Factorial function in Agora
func factorial(n) {
    if (n <= 1) {
        return 1;
    } else {
        return n * factorial(n - 1);
    }
}

// Test the factorial function
print("Factorial of 5: " + factorial(5));
print("Factorial of 10: " + factorial(10));
