# Fibonacci sequence in Potion
fib = (n):
  if (n <= 1):
    n
  else:
    fib(n - 1) + fib(n - 2)
  .

# Print first 10 Fibonacci numbers
1 to 10 do (i):
  fib(i) string print
  "\n" print
.
