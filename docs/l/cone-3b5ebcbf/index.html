fn fib(n u32) u32 {
    match n {
        0 => 1u,
        1 => 1u,
        _ => fib(n-2u) + fib(n-1u)
    }
}

fn main() {
    var i u32 = 0u;
    while i < 10u {
        console::write("{}\n", fib(i));
        i += 1u;
    }
}
