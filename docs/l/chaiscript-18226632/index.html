/*
Sieve of Eratosthenes
*/

def Sieve(n) {
  var a = range(0, n);
  a[1] = 0; // 1 is not prime

  var p = 2;

  while (p*p <= n) {
    var j = p*p;
    while (j <= n) {
      a[j] = 0;
      j += p;
    }

    p += 1;
    while (a[p] == 0) {
      p += 1;
    }
  }

  var primes = [];
  for (var i = 0; i < a.size(); ++i) {
    if (a[i] != 0) {
      primes.push_back(a[i]);
    }
  }
  return primes;
}

var primes = Sieve(100);
print(primes);
