begin
    integer procedure factorial(n);
        integer n;
    begin
        if n <= 1 then
            factorial := 1
        else
            factorial := n * factorial(n - 1)
    end;

    integer i;
    for i := 0 step 1 until 10 do
        outinteger(1, factorial(i))
end
