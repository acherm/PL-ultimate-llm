Sum of first n natural numbers

Origin · Download · manifest.json

module example
export sum, main
var sum : integer procedure (n : integer) integer
  var i, s : integer
  s := 0
  for i := 1 to n step 1 do
    s := s + i
  end for
  result s
end sum

var n : integer
get (n)
put (sum (n))
end main
end module
