Factorial Calculator

Origin · Download · manifest.json
code.imp · added: 2026-02-06T11:59:39Z
begin
  integer n, fact, i;

  n = 5;
  fact = 1;
  i = 1;

  while i <= n do
  begin
    fact = fact * i;
    i = i + 1
  end;

  print("Factorial of ", n, " is ", fact)
end
