public class Point {
  int x, y;
  public Point( int x, int y ) {
    this.x = x;
    this.y = y;
  }
  public String toString() {
    return "(" + x + ", " + y + ")";
  }
}

// Create a point
Point p = new Point( 7, 42 );

// Print it
print( p );

// Modify an instance variable
p.x = 9;

// Print it again
print( p );
