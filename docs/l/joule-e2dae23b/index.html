// Fibonacci sequence generator in Joule
define fib(n: int) : int {
    if (n <= 1) {
        return n;
    } else {
        return fib(n - 1) + fib(n - 2);
    }
}

// Main program
define main() {
    for i in 0..10 {
        print(fib(i));
    }
}
